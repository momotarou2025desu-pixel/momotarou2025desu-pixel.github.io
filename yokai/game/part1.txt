requency.setValueAtTime(Math.max(55, s.f1 * .50), t);
    sub.frequency.exponentialRampToValueAtTime(Math.max(55, s.f2 * .50), end);

    const subGain = audio.createGain();
    subGain.gain.value = s.f1 > 1200 ? .035 : .10;
    sub.connect(subGain);
    subGain.connect(filter);

    const vib = audio.createOscillator();
    vib.type = "sine";
    vib.frequency.value = s.wr;
    const vibGain = audio.createGain();
    vibGain.gain.value = s.wob;
    vib.connect(vibGain);
    vibGain.connect(osc.frequency);

    let vib2 = null;
    if (["rough","warble","tremble"].includes(s.shape)) {
      vib2 = audio.createOscillator();
      vib2.type = "triangle";
      vib2.frequency.value = s.shape === "warble" ? 4.5 : 8.5;
      const vib2Gain = audio.createGain();
      vib2Gain.gain.value = s.shape === "warble" ? 90 : 42;
      vib2.connect(vib2Gain);
      vib2Gain.connect(osc.frequency);
    }

    const ns = audio.createBufferSource();
    ns.buffer = makeNoiseBuffer(audio, s.dur);
    const noiseFilter = audio.createBiquadFilter();
    noiseFilter.type = "highpass";
    noiseFilter.frequency.value = s.f1 > 1000 ? 1800 : 520;
    const noiseGain = audio.createGain();
    noiseGain.gain.setValueAtTime(Math.max(.0001, s.noise * .82), t);
    noiseGain.gain.exponentialRampToValueAtTime(.0001, end);
    ns.connect(noiseFilter);
    noiseFilter.connect(noiseGain);
    noiseGain.connect(master);

    let dive = null;
    if (s.shape === "dive") {
      dive = audio.createOscillator();
      const diveGain = audio.createGain();
      dive.type = "sine";
      dive.frequency.value = 5.2;
      diveGain.gain.setValueAtTime(18, t);
      diveGain.gain.exponentialRampToValueAtTime(115, end);
      dive.connect(diveGain);
      diveGain.connect(osc.frequency);
    }

    osc.start(t);
    sub.start(t);
    vib.start(t);
    ns.start(t);
    if (vib2) vib2.start(t);
    if (dive) dive.start(t);

    osc.stop(end);
    sub.stop(end);
    vib.stop(end);
    ns.stop(end);
    if (vib2) vib2.stop(end);
    if (dive) dive.stop(end);

    sources.push(osc, sub, vib, ns);
    if (vib2) sources.push(vib2);
    if (dive) sources.push(dive);

    activeScreamGroups.push({ nodes: sources, endTime: end });
  }

  function terminalTone(audio, freq, start, dur, wave="square", gain=.08, endFreq=null) {
    const osc = audio.createOscillator();
    const g = audio.createGain();

    osc.type = wave;
    osc.frequency.setValueAtTime(freq, start);
    if (endFreq !== null) {
      osc.frequency.exponentialRampToValueAtTime(Math.max(35, endFreq), start + dur);
    }

    g.gain.setValueAtTime(.0001, start);
    g.gain.exponentialRampToValueAtTime(gain, start + .008);
    g.gain.setValueAtTime(gain, start + Math.max(.012, dur * .70));
    g.gain.exponentialRampToValueAtTime(.0001, start + dur);

    osc.connect(g);
    g.connect(audio.destination);
    osc.start(start);
    osc.stop(start + dur);
    terminalNodes.push(osc);
  }

  function terminalNoise(audio, start, dur, gain=.04, center=520) {
    const len = Math.max(1, Math.floor(audio.sampleRate * dur));
    const buf = audio.createBuffer(1, len, audio.sampleRate);
    const data = buf.getChannelData(0);
    let last = 0;

    for (let i = 0; i < len; i++) {
      const white = Math.random() * 2 - 1;
      last = last * .72 + white * .28;
      data[i] = last;
    }

    const src = audio.createBufferSource();
    const filter = audio.createBiquadFilter();
    const g = audio.createGain();

    src.buffer = buf;
    filter.type = "bandpass";
    filter.frequency.value = center;
    filter.Q.value = .8;

    g.gain.setValueAtTime(gain, start);
    g.gain.exponentialRampToValueAtTime(.0001, start + dur);

    src.connect(filter);
    filter.connect(g);
    g.connect(audio.destination);
    src.start(start);
    src.stop(start + dur);
    terminalNodes.push(src);
  }

  function isMobileGameLayout() {
    return window.matchMedia("(max-width: 600px), (pointer: coarse)").matches;
  }

  // 画面内の操作案内は端末に合わせて短く切り替える。
  function primaryActionWord() {
    return isMobileGameLayout() ? "タップ" : "クリック";
  }

  function syncCanvasLogicalSize() {
    if (!isMobileGameLayout()) {
      if (canvas.width !== 540 || canvas.height !== 960) {
        canvas.width = 540;
        canvas.height = 960;
        W = 540;
        H = 960;
      }
      return;
    }

    const rect = canvas.getBoundingClientRect();
    if (rect.width < 10 || rect.height < 10) return;

    // 横方向の論理解像度は540で固定し、縦だけ端末の見える比率に合わせる。
    // CSSの引き伸ばしではなくCanvas自体の比率を合わせるので絵が歪まない。
    const nextW = 540;
    const nextH = Math.max(560, Math.round(nextW * rect.height / rect.width));

    if (canvas.width !== nextW || canvas.height !== nextH) {
      canvas.width = nextW;
      canvas.height = nextH;
      W = nextW;
      H = nextH;

      if (player) {
        player.x = Math.max(0, Math.min(W - player.w, player.x));
        player.y = H - 82;
      }
    }
  }

        : window.innerHeight;

    const unitTop = gameUnit.getBoundingClientRect().top;
