  function ensureAudio() {
    if (!window.AudioContext && !window.webkitAudioContext) return null;
    if (!audioCtx) {
      const AudioCtor = window.AudioContext || window.webkitAudioContext;
      audioCtx = new AudioCtor();
    }
    if (audioCtx.state === "suspended") {
      audioCtx.resume().catch(() => {});
    }
    return audioCtx;
  }

  function stopNodes(nodes) {
    for (const node of nodes) {
      try { node.stop(); } catch (_) {}
      try { node.disconnect(); } catch (_) {}
    }
  }

  function stopAllGameAudio() {
    for (const group of activeScreamGroups) stopNodes(group.nodes);
    activeScreamGroups = [];
    stopNodes(terminalNodes);
    terminalNodes = [];

  function makeNoiseBuffer(audio, duration) {
    const len = Math.max(1, Math.floor(audio.sampleRate * duration));
    const buf = audio.createBuffer(1, len, audio.sampleRate);
    const data = buf.getChannelData(0);
    let last = 0;

    for (let i = 0; i < len; i++) {
      const white = Math.random() * 2 - 1;
      last = last * .80 + white * .20;
      data[i] = last;
    }
    return buf;
  }

  function setScreamPitchShape(osc, s, t) {
    const end = t + s.dur;
    osc.frequency.cancelScheduledValues(t);
    osc.frequency.setValueAtTime(Math.max(55, s.f1), t);

    switch (s.shape) {
      case "riseCut":
        osc.frequency.exponentialRampToValueAtTime(Math.max(60, s.f2), t + s.dur * .72);
        osc.frequency.exponentialRampToValueAtTime(Math.max(55, s.f2 * .72), end);
        break;

      case "jump":
        osc.frequency.linearRampToValueAtTime(s.f1 * .82, t + s.dur * .18);
        osc.frequency.linearRampToValueAtTime(s.f2, t + s.dur * .38);
        osc.frequency.linearRampToValueAtTime(Math.max(60, s.f2 * .66), end);
        break;

      case "riseDrop":
        osc.frequency.linearRampToValueAtTime(s.f2, t + s.dur * .42);
        osc.frequency.exponentialRampToValueAtTime(Math.max(65, s.f1 * .42), end);
        break;

      case "zigzag":
        osc.frequency.setValueAtTime(s.f1, t);
        osc.frequency.linearRampToValueAtTime(s.f1 * 1.75, t + s.dur * .17);
        osc.frequency.linearRampToValueAtTime(s.f1 * .72, t + s.dur * .34);
        osc.frequency.linearRampToValueAtTime(s.f1 * 1.45, t + s.dur * .52);
        osc.frequency.linearRampToValueAtTime(s.f1 * .60, t + s.dur * .71);
        osc.frequency.exponentialRampToValueAtTime(Math.max(60, s.f2), end);
        break;

      case "warble":
        osc.frequency.linearRampToValueAtTime(s.f1 * 1.18, t + s.dur * .20);
        osc.frequency.linearRampToValueAtTime(s.f1 * .75, t + s.dur * .40);
        osc.frequency.linearRampToValueAtTime(s.f1, t + s.dur * .58);
        osc.frequency.exponentialRampToValueAtTime(Math.max(55, s.f2), end);
        break;

      case "crush":
        osc.frequency.linearRampToValueAtTime(s.f1 * .60, t + s.dur * .42);
        osc.frequency.linearRampToValueAtTime(s.f1 * 1.12, t + s.dur * .60);
        osc.frequency.exponentialRampToValueAtTime(Math.max(55, s.f2), end);
        break;

      default:
        osc.frequency.exponentialRampToValueAtTime(Math.max(55, s.f2), end);
        break;
    }
  }
