  function drawClearBackground(timestamp) {
    const pulse = (Math.sin(timestamp * 0.004) + 1) * 0.5;

    const grad = ctx.createRadialGradient(W / 2, H / 2, 20, W / 2, H / 2, 520);
    grad.addColorStop(0, pulse > .5 ? "#17a995" : "#119884");
    grad.addColorStop(.48, "#0b756d");
    grad.addColorStop(1, "#063b3f");
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // レトロな放射線。
    ctx.save();
    ctx.translate(W / 2, H / 2);
    for (let i = 0; i < 24; i++) {
      const a1 = (Math.PI * 2 / 24) * i;
      const a2 = a1 + Math.PI / 24;
      const length = 700;

      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.lineTo(Math.cos(a1) * length, Math.sin(a1) * length);
      ctx.lineTo(Math.cos(a2) * length, Math.sin(a2) * length);
      ctx.closePath();
      ctx.fillStyle = i % 2 === 0
        ? "rgba(255,255,255,0.075)"
        : "rgba(0,0,0,0.035)";
      ctx.fill();
    }
    ctx.restore();
  }

  function drawFailBackground(timestamp) {
    const pulse = (Math.sin(timestamp * 0.010) + 1) * 0.5;
    const grad = ctx.createLinearGradient(0, 0, 0, H);
    grad.addColorStop(0, pulse > .5 ? "#a91520" : "#8e1018");
    grad.addColorStop(.50, "#651016");
    grad.addColorStop(1, "#2a0508");

    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, W, H);

    // 横走査線ノイズ。
    for (let y = 0; y < H; y += 8) {
      ctx.fillStyle = y % 16 === 0
        ? "rgba(255,255,255,0.035)"
        : "rgba(0,0,0,0.09)";
      ctx.fillRect(0, y, W, 3);
    }

    // 少量のグリッチ帯。
    const shift = Math.round(Math.sin(timestamp * .025) * 18);
    ctx.fillStyle = "rgba(255,220,200,0.07)";
    ctx.fillRect(shift, 118, W, 7);
    ctx.fillRect(-shift, 372, W, 5);
  }

  function fitTerminalFont(text, maxWidth) {
    let size = 74;
    ctx.font = `900 ${size}px "Yu Gothic", "Hiragino Kaku Gothic ProN", sans-serif`;

    while (size > 34 && ctx.measureText(text).width > maxWidth) {
      size -= 2;
      ctx.font = `900 ${size}px "Yu Gothic", "Hiragino Kaku Gothic ProN", sans-serif`;
    }
    return size;
  }

  function drawTerminalSoundText(timestamp) {
    if (!terminalSoundText || !terminalTextStartAt) return;

    const elapsed = Math.max(0, timestamp - terminalTextStartAt);
    const pop = Math.min(1, elapsed / 170);
    const scale = pop < 1 ? .66 + .44 * pop : 1;
    const shakeX = (flowState === FLOW.GAMEOVER_MINIONS || flowState === FLOW.GAMEOVER_BOSS) ? Math.sin(timestamp * .075) * 3 : 0;

    ctx.save();
    ctx.translate(W / 2 + shakeX, H / 2 - 5);
    ctx.scale(scale, scale);
    ctx.textAlign = "center";
    ctx.textBaseline = "middle";

    const size = fitTerminalFont(terminalSoundText, W * .88);
    ctx.font = `900 ${size}px "Yu Gothic", "Hiragino Kaku Gothic ProN", sans-serif`;
    ctx.lineJoin = "round";
    ctx.lineWidth = Math.max(7, size * .12);

    if (flowState === FLOW.CLEAR) {
      ctx.strokeStyle = "#06433f";
      ctx.fillStyle = "#fff36d";
      ctx.shadowColor = "rgba(255,255,255,.42)";
      ctx.shadowBlur = 18;
    } else {
      ctx.strokeStyle = "#3a0307";
      ctx.fillStyle = "#fff0c8";
      ctx.shadowColor = "rgba(255,70,50,.42)";
      ctx.shadowBlur = 16;
    }

    ctx.strokeText(terminalSoundText, 0, 0);
    ctx.fillText(terminalSoundText, 0, 0);
    ctx.restore();
  }

  function draw(timestamp) {
    ctx.clearRect(0, 0, W, H);

    if (flowState === FLOW.TITLE) {
      drawTitleScreen(timestamp);
      return;
    }

    if (flowState === FLOW.BOSS_READY) {
      drawBossReadyScreen(timestamp);
      return;
    }

    if (flowState === FLOW.CLEAR) {
      drawClearBackground(timestamp);

      // 最後の目だま妖怪の悲鳴文字は、クリア演出に切り替わるまで残す。
      if (!terminalSoundText) drawFloatingTexts(timestamp);

      drawTerminalSoundText(timestamp);
      drawRestartHint(timestamp);
      return;
    }

    if (flowState === FLOW.GAMEOVER_MINIONS || flowState === FLOW.GAMEOVER_BOSS) {
      drawFailBackground(timestamp);
      drawTerminalSoundText(timestamp);
      drawRestartHint(timestamp);
      return;
    }

    drawNormalBackground();
    drawPlayer(timestamp);

    if (flowState === FLOW.MINIONS) {
      enemies.forEach(e => {
        if (e.alive) drawEnemy(e, timestamp);
      });
    } else if (flowState === FLOW.BOSS) {
      drawBoss(timestamp);
      drawBossHp();
    }

    // お札弾
    bullets.forEach(b => {
      ctx.fillStyle = "#ffe7a4";
      ctx.fillRect(b.x - 1, b.y, b.w + 2, b.h);
      ctx.fillStyle = "#b42323";
      ctx.fillRect(b.x + 1, b.y + 3, b.w - 2, 3);
    });

    // 妖怪弾
    enemyBullets.forEach(b => {
      if (flowState === FLOW.BOSS) {
        ctx.fillStyle = "rgba(39,196,255,.28)";
        ctx.fillRect(b.x - 3, b.y - 2, b.w + 6, b.h + 4);
      }
      ctx.fillStyle = "#27c4ff";
      ctx.fillRect(b.x, b.y, b.w, b.h);
      ctx.fillStyle = "#d6ffff";
      ctx.fillRect(b.x + 2, b.y + 2, 2, 5);
    });

    drawFloatingTexts(timestamp);
  }

  function loop(timestamp) {
    update(timestamp);
    draw(timestamp);
    animationId = requestAnimationFrame(loop);
  }

  function getTouchPadX(e) {
    const rect = touchPad.getBoundingClientRect();
    return (e.clientX - rect.left) * (canvas.width / rect.width);
  }

  // GAME OVER後の5秒以内に操作された場合は、初期画面を挟まずその場で再戦する。
  function retryBossBattleImmediately({ preserveTouch = false } = {}) {
    const checkpointScore = bossCheckpointScore;
    const retryLife = bossStartLife();

    prepareStandaloneBattle({ preserveTouch });
    score = checkpointScore;
    bossCheckpointScore = checkpointScore;
    activateBossBattle(performance.now(), retryLife);
    startGameLoop();
  }
